const OPS_SS_ = '1pfeTChCtIV1RZKyjJRbjkXUL3xonTaJ0fsV0wBRbyZ0';
function proKey_(s) { return String(s || '').toLowerCase().replace(/[^a-z؀-ۿ]/g, ''); }
function proOps_(name, from, to) {
  const O = SpreadsheetApp.openById(OPS_SS_), tab = n => { const s = O.getSheetByName(n); if (!s) return []; const v = s.getDataRange().getValues(), h = v.shift().map(x => String(x).trim());
    return v.map(r => { const o = {}; h.forEach((k, i) => o[k] = r[i] instanceof Date ? Utilities.formatDate(r[i], TZ, 'yyyy-MM-dd HH:mm') : r[i]); return o; }); };
  const k = proKey_(name), inR = d => { d = String(d || '').slice(0, 10); return d >= from && d <= to; }, J = {}; tab('Jobs').forEach(j => J[j.JobNo] = j);
  const FROM = { Ibrahim: 'تحويل من دكتور إبراهيم', Custody: 'من العهدة', Employee: 'من جيبه', Owner: 'من إسماعيل', Credit: 'أجل' };
  const ex = tab('Expenses').filter(x => proKey_(x.By) === k && x.Status !== 'Void' && inR(x.Date)).map(x => { const j = J[x.JobNo];
    return { date: String(x.Date).slice(0, 10), at: String(x.At || '').slice(11, 16), jobNo: x.JobNo || '', car: j ? (j.Client || '') + ' — ' + [j.Make, j.Model].filter(Boolean).join(' ') : '', item: x.Item || '', supplier: x.Supplier || '', from: FROM[x.PaidFrom] || (x.Pay === 'Credit' ? 'أجل' : ''), amount: num_(x.Cost) }; });
  const oh = tab('Overheads').filter(o => proKey_(o.By) === k && /^Staff:/.test(String(o.Source)) && inR(o.Date)).map(o => ({ date: String(o.Date).slice(0, 10), at: String(o.At || '').slice(11, 16), jobNo: '', car: 'مصاريف عامة', item: o.Item || o.Sub || o.Cat || '', supplier: '', from: FROM[String(o.Source).split(':')[1]] || '', amount: num_(o.Amount) }));
  return ex.concat(oh);
}
function empPro(pin, empId, from, to) {
  const a = admin_(pin), e = rows_('Employees').find(r => r.ID === empId); if (!e) throw new Error('الموظف مش موجود');
  const ok = s => /^\d{4}-\d{2}-\d{2}$/.test(String(s || '')); to = ok(to) ? to : today_(); from = ok(from) ? from : to; if (from > to) { const x = from; from = to; to = x; }
  const S = settings_(), B = build_(from, to), inR = d => { d = String(d || '').slice(0, 10); return d >= from && d <= to; }, viewer = a.Role === 'Viewer';
  let spend = [], spendErr = ''; try { spend = proOps_(e.Name, from, to); } catch (x) { spendErr = String(x.message || x); }
  // money exactly like the payroll pages: the week pays base − late − absence − خصم + extra days + منح − سلفة; day 1 of the month pays Overtime + Bonus
  const monthly = e.Cycle === 'Monthly', noAtt = e.Mode === 'NoAttendance', otOk = e.Overtime === 'Yes' && !noAtt, otRate = num_(e.Salary) / num_(S.DayDivisor) / wh_(e, S);
  const prior = rows_('Attendance').filter(r => r.EmpID === e.ID && r.Date < from && r.Lng).sort((x, y) => num_(x.Epoch) - num_(y.Epoch));
  let city = prior.length ? cityOf_(prior[prior.length - 1]) : '';
  const rateFor = c => (c === 'Alex' && num_(e.SalaryAlex) ? num_(e.SalaryAlex) : num_(e.Salary)) / num_(S.DayDivisor);
  const dim = d => { const p = d.split('-'); return new Date(Date.UTC(+p[0], +p[1], 0)).getUTCDate(); };
  const days = range_(from, to).map(d => { const x = day_(e, d, B), c = cell_(e, x, B, d); if (x.city) city = x.city; const r = rateFor(city);
    const P = (B.att[e.ID + '|' + d] || []).filter(q => q.Type === 'in' || q.Type === 'out').sort((p, q) => num_(p.Epoch) - num_(q.Epoch)).map(q => ({ type: q.Type, time: q.Time, map: String(q.Map || '').split(' | ')[0], manual: String(q.Map || '').indexOf('Manual') === 0 }));
    const A = (B.adj[e.ID + '|' + d] || []).filter(q => q.ID).map(q => ({ id: q.ID, type: q.Type, amount: num_(q.Amount), reason: q.Reason || '', by: q.By || '' }));
    const m = { base: monthly ? num_(e.Salary) / dim(d) : r * (num_(S.WeekDays) || 7) / 7, late: (x.late ? num_(S.LateDeductionDays) * r : 0) + num_(x.lateManual), absent: x.absent ? num_(S.AbsenceDeductionDays) * r : 0,
      extra: x.extraDay ? num_(S.ExtraDayMultiplier) * r : 0, ot: otOk ? Math.max(0, x.overtime - num_(x.otDeducted)) * otRate : 0 };
    Object.keys(m).forEach(k => m[k] = r2_(m[k]));
    return Object.assign(c, { bonus: r2_(x.bonus), grant: r2_(x.grant), deduction: r2_(x.deduction), absent: !!x.absent, autoLate: !!x.late, extraDay: !!x.extraDay, leave: x.leave || '', isOff: !!x.isOff, future: d > today_(), city: city,
      otNet: r2_(Math.max(0, x.overtime - num_(x.otDeducted))), otDed: r2_(num_(x.otDeducted)), m: m, punches: P, adj: A, spend: spend.filter(s => s.date === d) }); });
  const sum = (L, f) => r2_(L.reduce((t, x) => t + num_(f(x)), 0)), adv = rows_('Advances');
  const advW = range_(from, to).filter(d => weekday_(d) === 'Saturday').map(w => ({ week: w, amount: r2_(advanceDue_(adv, e.ID, w)) })).filter(x => x.amount);
  const months = {}; range_(from, to).forEach(d => months[d.slice(0, 7)] = 1);
  const pays = rows_('Payments').filter(p => p.EmpID === e.ID && num_(p.Amount) && (p.Kind === 'Weekly' ? inR(p.Period) : months[String(p.Period).slice(0, 7)] || inR(p.PaidAt)))
    .map(p => ({ at: String(p.PaidAt).slice(0, 16), kind: p.Kind, period: p.Period, amount: num_(p.Amount), details: p.Details || '', by: p.By || '' }));
  const T = { base: sum(days, d => d.m.base), late: sum(days, d => d.m.late), absent: sum(days, d => d.m.absent), extra: sum(days, d => d.m.extra), ded: sum(days, d => d.deduction), grant: sum(days, d => d.grant),
    adv: sum(advW, x => x.amount), bonus: sum(days, d => d.bonus), ot: sum(days, d => d.m.ot), otH: sum(days, d => d.otNet) };
  T.week = r2_(T.base - T.late - T.absent - T.ded + T.extra + T.grant - T.adv); T.month = r2_(T.ot + T.bonus); T.total = r2_(T.week + T.month);
  T.paidW = sum(pays.filter(p => p.kind === 'Weekly' || p.kind === 'Monthly'), p => p.amount); T.paidM = sum(pays.filter(p => p.kind === 'Overtime'), p => p.amount); T.paid = r2_(T.paidW + T.paidM);
  if (viewer) Object.keys(T).forEach(k => T[k] = null);
  const reqs = rows_('Requests').filter(r => r.EmpID === e.ID).slice(-10).reverse().map(r => ({ type: r.Type, from: r.From, to: r.To, days: r.Days, status: r.Status || 'Pending', reason: r.Reason || '' }));
  const advs = adv.filter(r => r.EmpID === e.ID).slice(-10).reverse().map(r => ({ date: r.Date, amount: num_(r.Amount), inst: num_(r.Installment), reason: r.Reason || '' }));
  return { emp: { id: e.ID, name: e.Name, job: e.Job || '', phone: e.Phone || '', dayOff: e.DayOff || '', start: lat_(e, S), end: e.EndTime || '', hours: wh_(e, S), mode: e.Mode || '', cycle: monthly ? 'Monthly' : 'Weekly', ot: otOk, otRate: viewer ? null : r2_(otRate),
      salary: viewer ? '' : num_(e.Salary), daily: viewer ? null : r2_(rateFor('Cairo')), hire: e.HireDate || '', notes: e.Notes || '' },
    from: from, to: to, today: today_(), canEdit: !viewer, days: days, spendErr: spendErr, pays: pays, advs: advs, advW: advW, reqs: reqs, bal: balances_(e), T: T,
    set: { lateDays: num_(S.LateDeductionDays), absDays: num_(S.AbsenceDeductionDays), extraX: num_(S.ExtraDayMultiplier), weekDays: num_(S.WeekDays) || 7, divisor: num_(S.DayDivisor), otApproval: String(S.OTApproval || '').toLowerCase() === 'yes' } };
}
